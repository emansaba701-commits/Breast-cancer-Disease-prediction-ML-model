        "app_code = r'''\n",
        "from flask import Flask, render_template, request\n",
        "import numpy as np\n",
        "import pickle\n",
        "\n",
        "app = Flask(__name__)\n",
        "\n",
        "# Load model\n",
        "model = pickle.load(open(\"model.pkl\", \"rb\"))\n",
        "\n",
        "# Load scaler\n",
        "scaler = pickle.load(open(\"scaler.pkl\", \"rb\"))\n",
        "\n",
        "\n",
        "@app.route(\"/\")\n",
        "def home():\n",
        "    return render_template(\"index.html\")\n",
        "\n",
        "\n",
        "@app.route(\"/predict\", methods=[\"POST\"])\n",
        "def predict():\n",
        "\n",
        "    try:\n",
        "\n",
        "        # Get input\n",
        "        input_text = request.form[\"features\"]\n",
        "\n",
        "        # Convert comma-separated values into numbers\n",
        "        values = [\n",
        "            float(x.strip())\n",
        "            for x in input_text.split(\",\")\n",
        "            if x.strip()\n",
        "        ]\n",
        "\n",
        "        # Check number of features\n",
        "        if len(values) != 30:\n",
        "\n",
        "            return render_template(\n",
        "                \"index.html\",\n",
        "                error=f\"Expected 30 features, but received {len(values)} features.\"\n",
        "            )\n",
        "\n",
        "        # Convert to numpy\n",
        "        input_data = np.array(values).reshape(1, -1)\n",
        "\n",
        "        # Apply StandardScaler\n",
        "        input_scaled = scaler.transform(input_data)\n",
        "\n",
        "        # Prediction\n",
        "        prediction = model.predict(input_scaled)[0]\n",
        "\n",
        "        if prediction == 1:\n",
        "            result = \"Malignant_cancerous\"\n",
        "        else:\n",
        "            result = \"Benign_Non-cancerous\"\n",
        "\n",
        "        return render_template(\n",
        "            \"index.html\",\n",
        "            prediction=result\n",
        "        )\n",
        "\n",
        "    except Exception as e:\n",
        "\n",
        "        return render_template(\n",
        "            \"index.html\",\n",
        "            error=str(e)\n",
        "        )\n",
        "\n",
        "\n",
        "if __name__ == \"__main__\":\n",
        "\n",
        "    app.run(\n",
        "        host=\"0.0.0.0\",\n",
        "        port=5000,\n",
        "        debug=False\n",
        "    )\n",
